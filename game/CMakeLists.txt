# p3fes_game: hand-decompiled game code (Phase 5, GOAL.md). Function names equal the IDA names in
# symbols/p3fes.sym. The recompiled program's entry point of each replaced function
# (`p3fes::recomp_gen::fn_<name>_<addr>`, which generated call sites and the dispatch table use) is a one-line
# forwarder generated from game/replaced.txt (forwarders.cmake, below: p3fes_game_forwarders for the game, a
# stand-in hooked copy for p3fes_game_tests); game/forwarders.txt lists the functions whose entry point a source
# here defines itself (typed shims). The emitter must be run with `--replace game/replaced.txt` (recomp/README.md
# "Replacements"), otherwise the generated body and the hand-written one both define the symbol and the link of
# p3fes fails -- on purpose.
#
# The library depends on headers only (EEContext, EEMemory, the rt adapter API); the executables that
# link it supply p3fes::rt (p3fes_rt, or the aborting stubs in the recompiler's link check).

# The replaced functions' recomp_gen entry points (recomp/README.md "Forwarders"), generated from replaced.txt and
# forwarders.txt by forwarders.cmake: forwarders_game.cpp (plain one-line forwarders) is p3fes_game_forwarders, which
# p3fes_program and the link check link; forwarders_test.cpp (each forwarder asks the installed test stand-ins first)
# is compiled into p3fes_game_tests instead (tests/game/CMakeLists.txt), which links p3fes_game without the plain ones.
# So every executable holds exactly one definition of each entry point. The outputs are rewritten only when their
# text changes (restat byproducts of the stamp).
set(_fwd_dir ${CMAKE_CURRENT_BINARY_DIR}/forwarders)
set(P3FES_GAME_FORWARDERS_TEST ${_fwd_dir}/forwarders_test.cpp CACHE INTERNAL "the stand-in hooked forwarders (p3fes_game_tests)")
add_custom_command(
    OUTPUT ${_fwd_dir}/forwarders.stamp
    BYPRODUCTS ${_fwd_dir}/forwarders_game.cpp ${_fwd_dir}/forwarders_test.cpp
    COMMAND ${CMAKE_COMMAND} -E make_directory ${_fwd_dir}
    COMMAND ${CMAKE_COMMAND} -DREPLACED=${CMAKE_CURRENT_SOURCE_DIR}/replaced.txt -DLIST=${CMAKE_CURRENT_SOURCE_DIR}/forwarders.txt
            -DOUT_GAME=${_fwd_dir}/forwarders_game.cpp -DOUT_TEST=${_fwd_dir}/forwarders_test.cpp
            -P ${CMAKE_CURRENT_SOURCE_DIR}/forwarders.cmake
    COMMAND ${CMAKE_COMMAND} -E touch ${_fwd_dir}/forwarders.stamp
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/replaced.txt ${CMAKE_CURRENT_SOURCE_DIR}/forwarders.txt
            ${CMAKE_CURRENT_SOURCE_DIR}/forwarders.cmake
    COMMENT "game: the replaced functions' recomp_gen forwarders (replaced.txt, forwarders.txt)"
    VERBATIM)
add_custom_target(p3fes_game_forwarders_gen DEPENDS ${_fwd_dir}/forwarders.stamp)
add_library(p3fes_game_forwarders STATIC ${_fwd_dir}/forwarders_game.cpp forwarders.txt forwarders.cmake)
add_dependencies(p3fes_game_forwarders p3fes_game_forwarders_gen)
target_link_libraries(p3fes_game_forwarders PUBLIC p3fes_game)
# ~11,000 functions in one unit: more sections than the default COFF limit
target_compile_options(p3fes_game_forwarders PRIVATE /bigobj)
